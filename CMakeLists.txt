cmake_minimum_required(VERSION 3.20)
project(NethLocalWorker LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)

if(NOT NETH_LLAMA_SOURCE)
    message(FATAL_ERROR "Set NETH_LLAMA_SOURCE to the prepared llama.cpp tree.")
endif()

add_subdirectory("${NETH_LLAMA_SOURCE}" llama)
add_executable(neth-qwen-worker Windows/Stage.cpp)
target_link_libraries(neth-qwen-worker PRIVATE llama)

option(NETH_BUILD_CHECKS "Build the Qwen split equivalence check" OFF)
if(NETH_BUILD_CHECKS)
    add_executable(check-qwen-split scripts/check-qwen-split.cpp)
    target_link_libraries(check-qwen-split PRIVATE llama)
endif()
